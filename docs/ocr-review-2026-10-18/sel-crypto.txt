def _encode_integer(n: int | bytes) -> bytes:
    if isinstance(n, int):
        if n < 0:
            raise ValueError("Negative integers are not supported in certificate DER")
        b = n.to_bytes(max(1, (n.bit_length() + 7) // 8), 'big')
