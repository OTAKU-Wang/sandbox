   123	function CopyButton({ text, label = '复制' }: { text: string; label?: string }) {
   124	  const [copied, setCopied] = useState(false);
   125	  const handleCopy = async () => {
   126	    const ok = await copyText(text);
   127	    if (ok) {
   128	      setCopied(true);
   129	      window.setTimeout(() => setCopied(false), 2000);
   130	    } else {
   131	      message.error('复制失败，请手动选择文本复制');
   132	    }
   133	  };
   134	  return (
   135	    <Button size="small" icon={<CopyOutlined />} onClick={() => { void handleCopy(); }}>
   136	      {copied ? '已复制' : label}
   137	    </Button>
   138	  );
   139	}

   168	async function describeDownloadError(error: unknown): Promise<string> {
   169	  const data = (error as { response?: { data?: unknown } } | null)?.response?.data;
   170	  if (data instanceof Blob) {
   171	    try {
   172	      return describeFlowError(JSON.parse(await data.text()), '成果下载失败');
   173	    } catch {
   174	      return '成果下载失败：后端返回了无法解析的错误内容';
   175	    }
   176	  }
   177	  return describeFlowError(error, '成果下载失败');
   178	}

   180	async function saveBundleFile(bundleId: string, blob: Blob) {
   181	  const url = URL.createObjectURL(blob);
   182	  const anchor = document.createElement('a');
   183	  anchor.href = url;
   184	  anchor.download = `cds-bundle-${bundleId}.zip`;
   185	  document.body.appendChild(anchor);
   186	  anchor.click();
   187	  document.body.removeChild(anchor);
   188	  URL.revokeObjectURL(url);
   189	}

   463	function OpenApiPreview({ doc, endpoints }: { doc: OpenApiDoc | null; endpoints: ApiEndpoint[] }) {
   464	  const rows = flattenOpenApiOperations(doc);
   465	  const firstSlug = endpoints[0]?.slug;
   466	  const firstRow = rows[0];
   467	  const curl = firstSlug ? buildCurlExample(firstSlug, buildRequestExample(firstRow, endpoints[0])) : '';
   468	
   469	  if (!doc) {
   470	    return (
   471	      <Alert
   472	        type="info"
   473	        showIcon
   474	        message="该合约还没有可展示的 OpenAPI 文档"
   475	        description="端点创建后，后端会按合约与负载生成 OpenAPI 3.1 文档；未生成时此处保持空白，不做示例伪造。"
   476	      />
   477	    );
   478	  }
   479	
   480	  return (
   481	    <Space direction="vertical" size={12} style={{ width: '100%' }}>
   482	      <Descriptions size="small" column={3}>
   483	        <Descriptions.Item label="文档标题">{doc.info?.title ?? '后端未提供 info.title'}</Descriptions.Item>
   484	        <Descriptions.Item label="版本">{doc.info?.version ?? '-'}</Descriptions.Item>
   485	        <Descriptions.Item label="OpenAPI">{doc.openapi ?? '-'}</Descriptions.Item>
   486	      </Descriptions>
   487	
   488	      {rows.length ? (
   489	        <Table
   490	          rowKey="key"
   491	          size="small"
   492	          pagination={false}
   493	          dataSource={rows}
   494	          columns={[
   495	            { title: '方法', dataIndex: 'method', width: 80, render: (method: string) => <Tag color="blue">{method}</Tag> },
   496	            { title: '路径', dataIndex: 'path' },
   497	            { title: '说明', dataIndex: 'summary', render: (summary: string) => summary || <Text type="secondary">文档未提供 summary</Text> },
   498	            {
   499	              title: '参数',
   500	              dataIndex: 'parameters',
   501	              render: (parameters: string[]) =>
   502	                parameters.length ? parameters.join('，') : <Text type="secondary">无</Text>,
   503	            },
   504	            {
   505	              title: '请求体字段',
   506	              dataIndex: 'bodyFields',
   507	              render: (fields: { name: string; type: string; required: boolean }[], record) =>
   508	                fields.length ? (
   509	                  <Space size={4} wrap>
   510	                    {fields.map((field) => (
   511	                      <Tag key={field.name} color={field.required ? 'red' : 'default'}>
   512	                        {field.name}: {field.type}
   513	                      </Tag>
   514	                    ))}
   515	                  </Space>
   516	                ) : record.requestBodyRequired ? (
   517	                  <Text type="secondary">需请求体，但文档未声明字段</Text>
   518	                ) : (
   519	                  <Text type="secondary">无</Text>
   520	                ),
   521	            },
   522	            {
   523	              title: '响应码',
   524	              dataIndex: 'responseCodes',
   525	              width: 140,
   526	              render: (codes: string[]) => codes.join(' / ') || '-',
   527	            },
   528	          ]}
   529	        />
   530	      ) : (
   531	        <Text type="secondary">文档内没有任何接口定义</Text>
   532	      )}
   533	
   534	      {curl && (
   535	        <div>
   536	          <Space style={{ marginBottom: 6 }}>
   537	            <Text type="secondary" style={{ fontSize: 12 }}>
   538	              调用示例（外部路径固定 POST {openApiPathForSlug(firstSlug ?? '')}，使用平台 API Key 鉴权）
   539	            </Text>
   540	            <CopyButton text={curl} label="复制 curl" />
   541	          </Space>
   542	          <pre className="cds-json-block">{curl}</pre>
   543	        </div>
   544	      )}
   545	
   546	      <Collapse
   547	        size="small"
   548	        items={[
   549	          {
   550	            key: 'raw',
   551	            label: '查看后端返回的原始 OpenAPI JSON',
   552	            children: <pre className="cds-json-block">{JSON.stringify(doc, null, 2)}</pre>,
   553	          },
   554	        ]}
   555	      />
   556	    </Space>
   557	  );
   558	}

   602	/** apiUrl 两段都来自后端：合约文档 servers[0].url + 端点 path；文档没下发就只给路径，不编主机名。 */
   603	function endpointApiUrl(endpoint: ApiEndpoint, doc: OpenApiDoc | null) {
   604	  const path = endpoint.path ?? openApiPathForSlug(endpoint.slug);
   605	  const server = doc?.servers?.[0]?.url;
   606	  const base = typeof server === 'string' ? server.replace(/\/+$/, '') : '';
   607	  return { path, url: base ? `${base}${path}` : null };
   608	}

   921	async function createEndpointWithStatus(
   922	  create: (payload: ApiEndpointCreatePayload) => Promise<ApiEndpoint>,
   923	  update: (vars: { id: string; payload: ApiEndpointUpdatePayload }) => Promise<ApiEndpoint>,
   924	  payload: ApiEndpointCreatePayload,
   925	  online: boolean,
   926	) {
   927	  const endpoint = await create(payload);
   928	  if (online) return { endpoint, offlineWritten: true };
   929	  try {
   930	    return { endpoint, offlineWritten: (await update({ id: endpoint.id, payload: { status: 'disabled' } })).status !== 'enabled' };
   931	  } catch {
   932	    return { endpoint, offlineWritten: false };
   933	  }
   934	}

   939	  onCreated,
   940	}: {
   941	  contractId: string | null;

   943	  onCreated: () => void;
   944	}) {
   945	  const [form] = Form.useForm();

   947	  const [expiry, setExpiry] = useState('never');
   948	  const workloadsQuery = useWorkloads({ contractId, env: 'production', status: 'published' });
   949	  const createMutation = useCreateEndpoint();
   950	  const updateMutation = useUpdateEndpoint();
   951	  const queryClient = useQueryClient();
   952	  const workloads = (workloadsQuery.data?.items ?? []).filter((workload) => workload.contract_id);
   953	
   954	  const submit = async () => {
   955	    let values: { name: string; workload_id: string; description: string };
   956	    try {
   957	      values = await form.validateFields();
   958	    } catch {
   959	      return;
   960	    }
   961	    const workload = workloads.find((candidate) => candidate.id === values.workload_id);
   962	    const targetContract = workload?.contract_id ?? contractId;
   963	    if (!targetContract) {
   964	      message.error('关联任务未绑定合约，后端无法创建接口');
   965	      return;
   966	    }

   969	        (payload) => createMutation.mutateAsync(payload),

   971	        {

   974	          name: values.name.trim(),
   975	          description: values.description.trim(),

   977	        online,
   978	      );
   979	      await queryClient.invalidateQueries({ queryKey: flowKeys.openapi(targetContract) });
   980	      message.success(`接口已创建：${openApiPathForSlug(endpoint.slug)}`);
   981	      if (!offlineWritten) {

  1001	            </Col>
  1002	            <Col xs={24} md={8}>

  1004	                <Select
  1005	                  showSearch
  1006	                  optionFilterProp="label"
  1007	                  placeholder="请选择已发布的生产任务"
  1008	                  loading={workloadsQuery.isPending}
  1009	                  notFoundContent={workloadsQuery.isPending ? '加载中…' : '后端没有可绑定的已发布生产任务'}
  1010	                  options={workloads.map((workload) => ({ label: workload.name, value: workload.id }))}
  1011	                />

  1014	            <Col xs={24} md={8}>

  1265	function CallStatusTag({ call }: { call: ApiEndpointCall }) {
  1266	  if (!call.status) return <Tag>{UNKNOWN_TEXT}</Tag>;
  1267	  const base = statusMeta(CALL_STATUS_META, call.status);
  1268	  if (call.status === 'succeeded') return <Tag color={base.color}>{base.label}</Tag>;
  1269	  const known = call.code ? statusMeta(CALL_FAILURE_CODE_META, call.code) : null;
  1270	  const useCode = Boolean(call.code && known && known.label !== call.code);
  1271	  const tag = <Tag color={useCode ? known!.color : base.color}>{useCode ? known!.label : base.label}</Tag>;
  1272	  return call.message ? <Tooltip title={call.message}>{tag}</Tooltip> : tag;
  1273	}
