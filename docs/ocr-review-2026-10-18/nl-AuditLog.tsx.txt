    92	/** 操作结果列：图上没有胶囊，只有 13×12 实心图标 + 同色文字（成功 #00a870 / 失败 #e5484d）。 */
    93	function OutcomeCell({ outcome }: { outcome?: 'success' | 'failure' }) {
    94	  if (outcome === 'success') {
    95	    return <Text className="cds-audit-outcome is-success"><CheckCircleFilled />成功</Text>;
    96	  }
    97	  if (outcome === 'failure') {
    98	    return <Text className="cds-audit-outcome is-failure"><CloseCircleFilled />失败</Text>;
    99	  }
   100	  return <MissingField reason="后端按动作名派生操作结果（audit.py::audit_outcome）；本次返回未带该字段" />;
   101	}

   103	/** 4-2 的操作人：原型形如「张伟（沈阳数据中心R / 模型工程师）」，只拼库里真有的段。 */
   104	function OperatorText({ record }: { record: AuditRecordDetail }) {
   105	  if (!record.username) {
   106	    return <MissingField reason="该动作无登录用户上下文（系统自动锚定、节点上报等），库里 user_id 为空" />;
   107	  }
   108	  const roleLabel = record.role ? (ROLE_LABELS[record.role as keyof typeof ROLE_LABELS] ?? record.role) : null;
   109	  const parts = [record.organization || null, roleLabel].filter((part): part is string => !!part);
   110	  if (parts.length === 0) return <span>{record.username}</span>;
   111	  return <span>{record.username}<Text type="secondary">（{parts.join(' / ')}）</Text></span>;
   112	}

   114	function saveBlob(blob: Blob, filename: string) {
   115	  const url = URL.createObjectURL(blob);
   116	  const anchor = document.createElement('a');
   117	  anchor.href = url;
   118	  anchor.download = filename;
   119	  document.body.appendChild(anchor);
   120	  anchor.click();
   121	  document.body.removeChild(anchor);
   122	  URL.revokeObjectURL(url);
   123	}

   461	              <Descriptions.Item label="操作结果"><OutcomeCell outcome={detail.outcome} /></Descriptions.Item>
   462	              <Descriptions.Item label="操作对象" span={2}>
   463	                {auditActionLabel(detail)}
   464	                {detail.resource_id ? <> · <Text code>{detail.resource_id}</Text></> : <Text type="secondary"> · {UNKNOWN_TEXT}</Text>}
   465	              </Descriptions.Item>
   466	            </Descriptions>
   467	
   468	            <div className="cds-audit-section">操作内容</div>
   469	            {/* 图上这五项坐在一条 #f7f9fc 灰面板里（x23..696 y354..511，行距 28） */}
   470	            <div className="cds-audit-panel">
   471	              <Descriptions className="cds-audit-content" column={1}>
   472	                <Descriptions.Item label="操作类型"><Text code>{auditActionCodeText(detail)}</Text></Descriptions.Item>
   473	                {envText && <Descriptions.Item label="环境标识"><Text code>{envText}</Text></Descriptions.Item>}
   474	                {detail.related_contract && (
   475	                  <Descriptions.Item label="关联合约">
   476	                    {detail.related_contract.title}（编号 <Text code>{detail.related_contract.contract_no}</Text>）
   477	                  </Descriptions.Item>
   478	                )}
   479	                {linkedData && (
   480	                  <Descriptions.Item label="关联数据">{linkedData.join('、')}</Descriptions.Item>
   481	                )}
   482	                {durationText && <Descriptions.Item label="耗时">{durationText}</Descriptions.Item>}
   483	                <Descriptions.Item label="detail 原文">
   484	                  {/* 图外补充（转录 y25–613 无此块）：保留取证能力，但默认收起不抢版面 */}
   485	                  <Collapse
   486	                    size="small"
   487	                    items={[{
   488	                      key: 'raw',
   489	                      label: '展开查看 detail 原文（后端下发的 JSON，未做改写）',
   490	                      children: (
   491	                        <pre style={{ fontSize: 12, maxHeight: 180, overflow: 'auto', margin: 0 }}>
   492	                          {JSON.stringify(detail.detail, null, 2)}
   493	                        </pre>
   494	                      ),
   495	                    }]}
   496	                  />
   497	                </Descriptions.Item>
   498	              </Descriptions>
   499	            </div>
   500	            {optionalMissing.length > 0 && (
   501	              <Text type="secondary" style={{ fontSize: 12, display: 'block', marginTop: 8 }}>
   502	                本条审计未随附：{optionalMissing.join(' / ')}（audit_logs 无对应列，后端不下发即不显示）
   503	              </Text>
   504	            )}
   505	
   506	            <div className="cds-audit-section">不可篡改存证</div>
   507	            <Descriptions className="cds-audit-proof" column={1}>
   508	              <Descriptions.Item label="存证状态">
   509	                {detail.blockchain_tx_hash ? <Tag color="green">已锚定</Tag> : <Tag>未锚定</Tag>}
   510	              </Descriptions.Item>
   511	              <Descriptions.Item label="存证 TX">
   512	                {detail.blockchain_tx_hash
   513	                  ? <Text code>{detail.blockchain_tx_hash}</Text>
   514	                  : <MissingField reason="该记录还没有 blockchain_tx_hash；锚定后可用「存证验真」做 SM3 + Merkle 重算比对" />}
   515	              </Descriptions.Item>
   516	            </Descriptions>
   517	          </div>
   518	        )}
   519	      </Modal>
   520	
   521	      <AuditVerifyModal recordId={verifyTargetId} open={verifyOpen} onClose={() => setVerifyOpen(false)} />
   522	    </div>
   523	  );
