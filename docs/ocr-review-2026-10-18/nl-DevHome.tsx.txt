   126	 * （灰 #a6adb8 / 绿 #00a870），异常=感叹号圆 #e34d59；文字取 §4 正文 13px #1d2129。

   128	function StateText({ pill, icon }: { pill: StatePill; icon?: 'play' | 'check' | 'alert' }) {
   129	  const toneColor =
   130	    pill.tone === 'success'
   131	      ? 'var(--cds-success)'
   132	      : pill.tone === 'danger'
   133	        ? dangerInk
   134	        : pill.tone === 'warning'
   135	          ? 'var(--cds-orange)'
   136	          : '#a6adb8';
   137	  const glyph = icon ?? (pill.tone === 'danger' || pill.tone === 'warning' ? 'alert' : 'check');
   138	  const Icon = glyph === 'play' ? PlayCircleFilled : glyph === 'alert' ? ExclamationCircleFilled : CheckCircleFilled;
   139	  return (
   140	    <Text style={{ fontSize: 13, color: ink }}>
   141	      <Icon style={{ fontSize: 12, color: toneColor, marginRight: 8 }} />
   142	      {pill.label}
   143	    </Text>
   144	  );
   145	}
