   105	/** 会话目录 = 后端下发的 notebook_path 去掉文件名；开发环反代只允许列这个前缀以内。 */
   106	function sessionDirOf(notebookPath: string | null | undefined): string {
   107	  const segments = String(notebookPath ?? '').split('/').filter(Boolean);
   108	  segments.pop();
   109	  return segments.join('/');
   110	}

   112	function countFiles(nodes: WorkbenchFileNode[]): number {
   113	  return nodes.reduce(
   114	    (acc, node) => acc + (node.isDirectory ? countFiles(node.children ?? []) : 1),

   130	export default function NotebookWorkbench() {
   131	  const [searchParams, setSearchParams] = useSearchParams();
   132	  const taskParam = searchParams.get('task');
   133	  const [pickedId, setPickedId] = useState<string | null>(null);
   134	  const [createOpen, setCreateOpen] = useState(false);
   135	  const notebooksQuery = useNotebooks(null, 1, 50);
   136	  const items = notebooksQuery.data?.items ?? [];
   137	
   138	  // ?task=<id>：卡片跳进来的上下文。选中态是从「地址栏 + 后端列表」推导出来的，不走 effect：
   139	  // 只有列表里真的有这个 id 才算选中，否则 /notebooks?task=<删掉的本子> 会对不存在的 id
   140	  // 发详情请求，页面挂在 404 上而不是回到「还没选」。列表还在路上时先转圈，不闪一下空状态。
   141	  const taskKnown = !!taskParam && items.some((item) => item.id === taskParam);
   142	  const selectedId = pickedId ?? (taskKnown ? taskParam : null);
   143	  const awaitingTaskParam = pickedId === null && !!taskParam && !notebooksQuery.data;
   144	  const { data: detail, isLoading, isError, error, refetch } = useNotebook(selectedId);
   145	
   146	  // 地址栏里的 task 参数刻意留着（刷新/收藏都回到同一个本子），只在手动换本子时清掉。
   147	  const pick = useCallback(
   148	    (id: string | null) => {
   149	      setPickedId(id);
   150	      if (searchParams.has('task')) setSearchParams({}, { replace: true });
   151	    },
   152	    [searchParams, setSearchParams],
   153	  );
   154	
   155	  return (
   156	    <div className="nbw-root">
   157	      <div className="nbw-titlecard">
   158	        <div>
   159	          <div className="nbw-titlecard-title">{NBW_TITLE}</div>
   160	          <div className="nbw-titlecard-sub">{NBW_SUBTITLE}</div>
   161	        </div>
   162	        <BackButton />
   163	      </div>
   164	
   165	      {/* 原型上没有告警条；但后端真报「没有可用 MicroVM 运行时」时必须挂出来（honest degradation），
   166	          运行时健康时这个组件自己返回 null，画面回到原型样。 */}
   167	      <RuntimeStatusBanner />
   168	
   169	      {isError ? <QueryErrorAlert error={error} message="调试对象加载失败" onRetry={() => { void refetch(); }} /> : null}
   170	
   171	      {!selectedId ? (
   172	        awaitingTaskParam ? (
   173	          <CenteredLoading />
   174	        ) : (
   175	          <PickNotebookCard
   176	            items={items.map((item) => ({ id: item.id, title: item.title }))}
   177	            onPick={pick}
   178	            onCreate={() => setCreateOpen(true)}
   179	          />
   180	        )
   181	      ) : isLoading ? (
   182	        <CenteredLoading />
   183	      ) : detail ? (
   184	        <NotebookWorkspace
   185	          key={detail.id}
   186	          notebook={detail}
   187	          siblings={items.map((item) => ({ id: item.id, title: item.title, task_no: item.task_no }))}
   188	          onPick={pick}
   189	          onCreate={() => setCreateOpen(true)}
   190	        />
   191	      ) : null}
   192	
   193	      <CreateNotebookModal
   194	        open={createOpen}
   195	        onClose={() => setCreateOpen(false)}
   196	        onCreated={(nb) => {
   197	          setCreateOpen(false);
   198	          pick(nb.id);
   199	        }}
   200	      />
   201	    </div>
   202	  );
   203	}
