   559	        extra={grantedCount === null

   563	        {contractQuery.isError ? (

   581	            pagination={false}

   599

   601	      <Card
   602	        className="t24-card"
   603	        size="small"
   604	        title={
   605	          <span className="t24-ht">
   606	            <span className="t24-bar" />
   607	            <Tooltip title={RESULT_SET_SCOPE_HINT}>
   608	              <span>结果集</span>
   609	            </Tooltip>
   610	            <ExportOutlined className="t24-hicon" />
   611	            <span className="t24-hnote">{RESULT_SET_NOTE}</span>
   612	          </span>
   613	        }
   614	        extra={<Link to={`/deliveries?contract=${encodeURIComponent(contractId ?? '')}`}>前往我的结果集<span className="t24-arrow">&gt;</span></Link>}
   615	      >
   616	        <Table<DatasetBundleRow>
   617	          className="t24-table"
   618	          rowKey="bundle_id"
   619	          size="small"
   620	          columns={bundleColumns}
   621	          dataSource={bundles.data?.items ?? []}
   622	          loading={bundles.isPending}
   623	          pagination={false}
   624	          locale={tableEmpty(bundleTotal ? '已有成果包但本页未取到' : '该合约还没有产出成果包')}

   626	        {bundleTotal !== null && (bundles.data?.items.length ?? 0) < bundleTotal ? (

   629	            图上无分页器，差额部分不去翻页假装已列全。

   632	      </Card>
   633	
   634	      <Modal

   640	      >
   641	        {detailProduct?.product ? (
